CREATE TABLE IF NOT EXISTS s.t (
  id bigint GENERATED BY DEFAULT AS IDENTITY (START WITH 10) PRIMARY KEY,
  name varchar(100) NOT NULL DEFAULT '' COLLATE "C",
  ref int REFERENCES u (id) ON DELETE SET NULL ON UPDATE CASCADE DEFERRABLE INITIALLY DEFERRED,
  total numeric GENERATED ALWAYS AS (price * qty) STORED,
  key text,
  CONSTRAINT t_name_key UNIQUE NULLS NOT DISTINCT (name),
  CHECK (total >= 0),
  FOREIGN KEY (ref, period valid) REFERENCES u (id, period valid),
  EXCLUDE USING gist (during WITH &&),
  NOT NULL name,
  LIKE other INCLUDING ALL
);
