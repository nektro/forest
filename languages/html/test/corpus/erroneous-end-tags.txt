================================================================================
Paragraph end tag after a block closed it
================================================================================

<p>text<div>block</div></p>

--------------------------------------------------------------------------------

(document
  (element
    start: (start_tag
      name: (tag_name))
    (text))
  (element
    start: (start_tag
      name: (tag_name))
    (text)
    end: (end_tag
      name: (tag_name)))
  (erroneous_end_tag
    name: (erroneous_end_tag_name)))
